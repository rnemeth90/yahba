# YAHBA Web GUI Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add a `yahba web` subcommand that serves a browser UI to configure and run load tests, watch live charts, and browse and compare saved run history.

**Architecture:** A new `internal/web` Go package wraps the existing `worker.Work` engine (using its `progressChan` for per-request results), aggregates results into one-second `Sample`s, persists runs and samples in SQLite, and exposes a JSON + SSE API. A TypeScript SPA in `web/` is bundled by esbuild into `internal/web/static/` (committed) and embedded with `go:embed`.

**Tech Stack:** Go 1.24 (`net/http` ServeMux method patterns, `go:embed`), `modernc.org/sqlite` (pure Go), testify, TypeScript, esbuild, uPlot, vitest.

## Global Constraints

Copied from `docs/superpowers/specs/2026-11-20-web-gui-design.md`:

- `yahba web` binds to `127.0.0.1:8080` by default (`--addr` overrides).
- SQLite at `~/.yahba/history.db` via `modernc.org/sqlite` (pure Go, **no cgo**).
- TypeScript source in `web/`; esbuild via `make web`; the compiled bundle in `internal/web/static/` is **committed** so `go:embed` and `go install` work without Node.
- Charts use uPlot. Live updates use SSE.
- One active run at a time (second start → HTTP 409). Invalid config → HTTP 400 with the field name. Client disconnect does not stop a run. On startup, runs left `running` are marked `aborted`.
- Run statuses: `running`, `completed`, `stopped`, `aborted`, `failed`.
- API routes: `POST /api/runs`, `GET /api/runs/{id}/events`, `DELETE /api/runs/{id}`, `GET /api/runs`, `GET /api/runs/{id}`, `GET /api/compare?a=&b=`.
- Out of scope: authentication, multi-user, remote-exposure hardening, concurrent runs.

Repo conventions:

- `main.go` has unrelated uncommitted changes. **Never** `git add -A` or `git commit -a`; stage explicit paths only.
- Every commit message ends with the trailer `Co-authored-by: Copilot <223556219+Copilot@users.noreply.github.com>` (pass as a second `-m`).
- Go tests use `github.com/stretchr/testify` (`assert`, `require`). Format with `gofmt -w -s`.
- Frontend renders user-supplied strings only via text nodes (`textContent` / `append(string)`), never `innerHTML`.

```go
package web

import (
	"math"
	"sort"
	"time"

	"github.com/rnemeth90/yahba/internal/report"
)

// Sample is one second of aggregated results.
type Sample struct {
	T           int         `json:"t"`
	Count       int         `json:"count"`
	Errors      int         `json:"errors"`
	AvgMs       float64     `json:"avg_ms"`
	P50Ms       float64     `json:"p50_ms"`
	P95Ms       float64     `json:"p95_ms"`
	P99Ms       float64     `json:"p99_ms"`
	StatusCodes map[int]int `json:"status_codes"`
}

// Sampler groups results into one-second buckets keyed by completion time.
type Sampler struct {
	start time.Time
	cur   int
	lat   []time.Duration
	errs  int
	codes map[int]int
}

func NewSampler(start time.Time) *Sampler {
	return &Sampler{start: start, codes: map[int]int{}}
}

// isFailure matches report.Report.Failures: no response, or status >= 400.
func isFailure(code int) bool { return code == 0 || code >= 400 }

// Add records r and returns the previous second's Sample when r begins a later
// second. Results arriving for an earlier second are folded into the current one.
func (s *Sampler) Add(r report.Result) []Sample {
	sec := int(r.EndTime.Sub(s.start) / time.Second)
	var out []Sample
	if sec > s.cur {
		out = s.Flush()
		s.cur = sec
	}
	s.lat = append(s.lat, r.ElapsedTime)
	s.codes[r.ResultCode]++
	if isFailure(r.ResultCode) {
		s.errs++
	}
	return out
}

// Flush returns the in-progress second as a Sample and resets it; nil if empty.
func (s *Sampler) Flush() []Sample {
	if len(s.lat) == 0 {
		return nil
	}
	sorted := append([]time.Duration(nil), s.lat...)
	sort.Slice(sorted, func(i, j int) bool { return sorted[i] < sorted[j] })
	var total time.Duration
	for _, d := range sorted {
		total += d
	}
	smp := Sample{
		T:           s.cur,
		Count:       len(sorted),
		Errors:      s.errs,
		AvgMs:       ms(total / time.Duration(len(sorted))),
		P50Ms:       ms(percentile(sorted, 50)),
		P95Ms:       ms(percentile(sorted, 95)),
		P99Ms:       ms(percentile(sorted, 99)),
		StatusCodes: s.codes,
	}
	s.lat = s.lat[:0]
	s.errs = 0
	s.codes = map[int]int{}
	return []Sample{smp}
}

// percentile uses the nearest-rank method on an ascending slice.
func percentile(sorted []time.Duration, p float64) time.Duration {
	idx := int(math.Ceil(p/100*float64(len(sorted)))) - 1
	if idx < 0 {
		idx = 0
	}
	if idx >= len(sorted) {
		idx = len(sorted) - 1
	}
	return sorted[idx]
}

func ms(d time.Duration) float64 { return float64(d) / float64(time.Millisecond) }
```

```go
package web

import (
	"sort"
	"time"

	"github.com/rnemeth90/yahba/internal/report"
)

// Summary is the compact, numeric form of a finished run that is stored and
// compared. It deliberately omits the per-request results of report.Report.
type Summary struct {
	Host          string      `json:"host"`
	Method        string      `json:"method"`
	TotalRequests int         `json:"total_requests"`
	Successes     int         `json:"successes"`
	Failures      int         `json:"failures"`
	DurationMs    int64       `json:"duration_ms"`
	RPS           float64     `json:"rps"`
	MinMs         float64     `json:"min_ms"`
	AvgMs         float64     `json:"avg_ms"`
	P50Ms         float64     `json:"p50_ms"`
	P95Ms         float64     `json:"p95_ms"`
	P99Ms         float64     `json:"p99_ms"`
	MaxMs         float64     `json:"max_ms"`
	StatusCodes   map[int]int `json:"status_codes"`
	BytesSent     int         `json:"bytes_sent"`
	BytesReceived int         `json:"bytes_received"`
}

func Summarize(r report.Report) Summary {
	s := Summary{
		Host:          r.Host,
		Method:        r.Method,
		TotalRequests: r.TotalRequests,
		Successes:     r.Successes,
		Failures:      r.Failures,
		DurationMs:    r.Duration.Milliseconds(),
		StatusCodes:   r.StatusCodes,
		BytesSent:     r.Throughput.TotalBytesSent,
		BytesReceived: r.Throughput.TotalBytesReceived,
	}
	if r.Duration > 0 {
		s.RPS = float64(r.TotalRequests) / r.Duration.Seconds()
	}
	if len(r.Results) == 0 {
		return s
	}
	lat := make([]time.Duration, len(r.Results))
	var total time.Duration
	for i, res := range r.Results {
		lat[i] = res.ElapsedTime
		total += res.ElapsedTime
	}
	sort.Slice(lat, func(i, j int) bool { return lat[i] < lat[j] })
	s.MinMs = ms(lat[0])
	s.MaxMs = ms(lat[len(lat)-1])
	s.AvgMs = ms(total / time.Duration(len(lat)))
	s.P50Ms = ms(percentile(lat, 50))
	s.P95Ms = ms(percentile(lat, 95))
	s.P99Ms = ms(percentile(lat, 99))
	return s
}
```

```go
package web

import (
	"database/sql"
	"encoding/json"
	"errors"
	"time"

	_ "modernc.org/sqlite"
)

var ErrNotFound = errors.New("run not found")

type RunStatus string

const (
	StatusRunning   RunStatus = "running"
	StatusCompleted RunStatus = "completed"
	StatusStopped   RunStatus = "stopped"
	StatusAborted   RunStatus = "aborted"
	StatusFailed    RunStatus = "failed"
)

type Run struct {
	ID         int64           `json:"id"`
	StartedAt  time.Time       `json:"started_at"`
	DurationMs int64           `json:"duration_ms"`
	Status     RunStatus       `json:"status"`
	Config     json.RawMessage `json:"config"`
	Summary    json.RawMessage `json:"summary,omitempty"`
}

const schema = `
CREATE TABLE IF NOT EXISTS runs (
	id          INTEGER PRIMARY KEY AUTOINCREMENT,
	started_at  TEXT NOT NULL,
	duration_ms INTEGER NOT NULL DEFAULT 0,
	status      TEXT NOT NULL,
	config      TEXT NOT NULL,
	summary     TEXT
);
CREATE TABLE IF NOT EXISTS samples (
	run_id INTEGER NOT NULL REFERENCES runs(id) ON DELETE CASCADE,
	t      INTEGER NOT NULL,
	data   TEXT NOT NULL,
	PRIMARY KEY (run_id, t)
);`

// Store persists runs and their per-second samples in SQLite.
type Store struct {
	db *sql.DB
}

func OpenStore(path string) (*Store, error) {
	dsn := "file:" + path + "?_pragma=busy_timeout(5000)&_pragma=journal_mode(WAL)&_pragma=foreign_keys(1)"
	db, err := sql.Open("sqlite", dsn)
	if err != nil {
		return nil, err
	}
	db.SetMaxOpenConns(1)
	if _, err := db.Exec(schema); err != nil {
		db.Close()
		return nil, err
	}
	return &Store{db: db}, nil
}

func (s *Store) Close() error { return s.db.Close() }

func (s *Store) CreateRun(cfg json.RawMessage, startedAt time.Time) (int64, error) {
	res, err := s.db.Exec(
		`INSERT INTO runs (started_at, status, config) VALUES (?, ?, ?)`,
		startedAt.UTC().Format(time.RFC3339Nano), StatusRunning, string(cfg),
	)
	if err != nil {
		return 0, err
	}
	return res.LastInsertId()
}

func (s *Store) AddSample(runID int64, smp Sample) error {
	data, err := json.Marshal(smp)
	if err != nil {
		return err
	}
	_, err = s.db.Exec(`INSERT OR REPLACE INTO samples (run_id, t, data) VALUES (?, ?, ?)`, runID, smp.T, string(data))
	return err
}

func (s *Store) FinishRun(id int64, status RunStatus, durationMs int64, summary json.RawMessage) error {
	var sum any
	if len(summary) > 0 {
		sum = string(summary)
	}
	_, err := s.db.Exec(`UPDATE runs SET status = ?, duration_ms = ?, summary = ? WHERE id = ?`, status, durationMs, sum, id)
	return err
}

func (s *Store) GetRun(id int64) (Run, error) {
	row := s.db.QueryRow(`SELECT id, started_at, duration_ms, status, config, summary FROM runs WHERE id = ?`, id)
	run, err := scanRun(row)
	if errors.Is(err, sql.ErrNoRows) {
		return Run{}, ErrNotFound
	}
	return run, err
}

func (s *Store) ListRuns() ([]Run, error) {
	rows, err := s.db.Query(`SELECT id, started_at, duration_ms, status, config, summary FROM runs ORDER BY id DESC`)
	if err != nil {
		return nil, err
	}
	defer rows.Close()
	runs := []Run{}
	for rows.Next() {
		run, err := scanRun(rows)
		if err != nil {
			return nil, err
		}
		runs = append(runs, run)
	}
	return runs, rows.Err()
}

func (s *Store) Samples(runID int64) ([]Sample, error) {
	rows, err := s.db.Query(`SELECT data FROM samples WHERE run_id = ? ORDER BY t`, runID)
	if err != nil {
		return nil, err
	}
	defer rows.Close()
	out := []Sample{}
	for rows.Next() {
		var data string
		if err := rows.Scan(&data); err != nil {
			return nil, err
		}
		var smp Sample
		if err := json.Unmarshal([]byte(data), &smp); err != nil {
			return nil, err
		}
		out = append(out, smp)
	}
	return out, rows.Err()
}

// MarkAbandoned flags runs left `running` by a crash or kill as aborted.
func (s *Store) MarkAbandoned() error {
	_, err := s.db.Exec(`UPDATE runs SET status = ? WHERE status = ?`, StatusAborted, StatusRunning)
	return err
}

type rowScanner interface{ Scan(dest ...any) error }

func scanRun(r rowScanner) (Run, error) {
	var (
		run     Run
		started string
		cfg     string
		summary sql.NullString
	)
	if err := r.Scan(&run.ID, &started, &run.DurationMs, &run.Status, &cfg, &summary); err != nil {
		return Run{}, err
	}
	t, err := time.Parse(time.RFC3339Nano, started)
```

```go
// maxRequests bounds the buffered result channels allocated per run.
const maxRequests = 1_000_000

// RunRequest is the subset of run options exposed in the web form. Proxy
// credentials are intentionally excluded because the config is stored in history.
type RunRequest struct {
	URL             string `json:"url"`
	Method          string `json:"method"`
	Requests        int    `json:"requests"`
	RPS             int    `json:"rps"`
	Workers         int    `json:"workers"`
	Timeout         int    `json:"timeout"`
	Headers         string `json:"headers"`
	Body            string `json:"body"`
	Insecure        bool   `json:"insecure"`
	KeepAlive       bool   `json:"keep_alive"`
	HTTP2           bool   `json:"http2"`
	Compression     bool   `json:"compression"`
	RandomUserAgent bool   `json:"random_user_agent"`
}

// ValidationError names the offending form field so the UI can highlight it.
type ValidationError struct {
	Field   string
	Message string
}

func (e *ValidationError) Error() string { return e.Message }

var fieldByError = map[error]string{
	config.ErrMissingHost:             "url",
	config.ErrInvalidProtocolScheme:   "url",
	config.ErrInvalidHost:             "url",
	config.ErrInvalidIPAddressForHost: "url",
	config.ErrInvalidMethod:           "method",
	config.ErrMissingBody:             "body",
	config.ErrInvalidRequests:         "requests",
	config.ErrInvalidTimeout:          "timeout",
	config.ErrInvalidRPS:              "rps",
	config.ErrInvalidHeaders:          "headers",
	config.ErrInvalidHTTPConfig:       "http2",
}

func fieldFor(err error) string {
	for target, field := range fieldByError {
		if errors.Is(err, target) {
			return field
		}
	}
	return ""
}

func (r RunRequest) ToConfig() (config.Config, error) {
	if r.Requests > maxRequests {
		return config.Config{}, &ValidationError{"requests", fmt.Sprintf("requests must be at most %d", maxRequests)}
	}
	if r.Workers <= 0 {
		return config.Config{}, &ValidationError{"workers", "workers must be greater than 0"}
	}

	method := strings.ToUpper(strings.TrimSpace(r.Method))
	if method == "" {
		method = "GET"
	}

	cfg := config.Config{
		URL:             strings.TrimSpace(r.URL),
		Method:          method,
		Requests:        r.Requests,
		RPS:             r.RPS,
		Workers:         r.Workers,
		Timeout:         r.Timeout,
		Headers:         r.Headers,
		Body:            r.Body,
		Insecure:        r.Insecure,
		KeepAlive:       r.KeepAlive,
		HTTP2:           r.HTTP2,
		Compression:     r.Compression,
		RandomUserAgent: r.RandomUserAgent,
		LogLevel:        "error",
		OutputFormat:    "raw",
		OutputFile:      "stdout",
		Silent:          true,
		Logger:          logger.New("error", "stdout", true),
	}

	if err := cfg.Validate(); err != nil {
		return config.Config{}, &ValidationError{Field: fieldFor(err), Message: err.Error()}
	}
	if cfg.Headers != "" {
		parsed, err := util.ParseHeaders(cfg.Headers)
		if err != nil {
			return config.Config{}, &ValidationError{"headers", err.Error()}
		}
		cfg.ParsedHeaders = parsed
	}
	return cfg, nil
}
```

```go
package web

import (
	"context"
	"encoding/json"
	"errors"
	"log"
	"net/http"
	"sync"
	"sync/atomic"
	"time"

	"github.com/rnemeth90/yahba/internal/config"
	"github.com/rnemeth90/yahba/internal/report"
	"github.com/rnemeth90/yahba/internal/worker"
)

var ErrBusy = errors.New("a run is already active")

type activeRun struct {
	id      int64
	cancel  context.CancelFunc
	stopped atomic.Bool
	samples []Sample
	subs    map[chan Sample]struct{}
	done    chan struct{}
}

// Runner executes at most one load test at a time, persisting its samples and
// final summary and fanning live samples out to subscribers.
type Runner struct {
	store  *Store
	mu     sync.Mutex
	active *activeRun
}

func NewRunner(store *Store) *Runner { return &Runner{store: store} }

func (r *Runner) Start(req RunRequest) (int64, error) {
	cfg, err := req.ToConfig()
	if err != nil {
		return 0, err
	}
	cfgJSON, err := json.Marshal(req)
	if err != nil {
		return 0, err
	}

	r.mu.Lock()
	defer r.mu.Unlock()
	if r.active != nil {
		return 0, ErrBusy
	}
	started := time.Now()
	id, err := r.store.CreateRun(cfgJSON, started)
	if err != nil {
		return 0, err
	}
	ctx, cancel := context.WithCancel(context.Background())
	a := &activeRun{id: id, cancel: cancel, subs: map[chan Sample]struct{}{}, done: make(chan struct{})}
	r.active = a
	go r.execute(ctx, a, cfg, started)
	return id, nil
}

func (r *Runner) execute(ctx context.Context, a *activeRun, cfg config.Config, started time.Time) {
	defer close(a.done)
	defer a.cancel()

	jobs := make([]worker.Job, cfg.Requests)
	for i := range jobs {
		jobs[i] = worker.Job{ID: i, Host: cfg.URL, Method: cfg.Method, Body: cfg.Body}
	}
	factory := func(id int, jobChan <-chan worker.Job, resultChan chan<- report.Result, client *http.Client, c config.Config) worker.Worker {
		return *worker.NewWorker(id, jobChan, resultChan, client, c)
	}

	progress := make(chan report.Result, cfg.Requests)
	reportChan := make(chan report.Report, 1)
	workDone := make(chan struct{})
	go func() {
		defer close(workDone)
		worker.Work(ctx, cfg, jobs, reportChan, factory, progress)
	}()

	sampler := NewSampler(started)
	handle := func(res report.Result) {
		for _, s := range sampler.Add(res) {
			r.publish(a, s)
		}
	}

	// worker.Work returns without closing its channels if the HTTP client cannot
	// be built, so also watch workDone and drain whatever is buffered.
loop:
	for {
		select {
		case res, ok := <-progress:
			if !ok {
				break loop
			}
			handle(res)
		case <-workDone:
			for {
				select {
				case res, ok := <-progress:
					if !ok {
						break loop
					}
					handle(res)
				default:
					break loop
				}
			}
		}
	}
	for _, s := range sampler.Flush() {
		r.publish(a, s)
	}

	status := StatusFailed
	var summary json.RawMessage
	select {
	case rep, ok := <-reportChan:
		if ok {
			status = StatusCompleted
			if a.stopped.Load() {
				status = StatusStopped
			}
			summary, _ = json.Marshal(Summarize(rep))
		}
	default:
	}

	if err := r.store.FinishRun(a.id, status, time.Since(started).Milliseconds(), summary); err != nil {
		log.Printf("web: failed to finish run %d: %v", a.id, err)
	}

	r.mu.Lock()
	for ch := range a.subs {
		close(ch)
	}
	a.subs = nil
	r.active = nil
	r.mu.Unlock()
}

func (r *Runner) publish(a *activeRun, s Sample) {
	if err := r.store.AddSample(a.id, s); err != nil {
		log.Printf("web: failed to save sample for run %d: %v", a.id, err)
	}
	r.mu.Lock()
	defer r.mu.Unlock()
	a.samples = append(a.samples, s)
	for ch := range a.subs {
		select {
		case ch <- s:
		default: // slow subscriber; it can reconnect and replay
		}
	}
}

// Stop cancels the active run if it has the given id.
func (r *Runner) Stop(id int64) bool {
	r.mu.Lock()
	a := r.active
	r.mu.Unlock()
	if a == nil || a.id != id {
		return false
	}
	a.stopped.Store(true)
	a.cancel()
	return true
}

// Subscribe returns the samples produced so far plus a channel of new ones. The
// channel is closed when the run finishes. ok is false if id is not active.
func (r *Runner) Subscribe(id int64) (past []Sample, ch <-chan Sample, unsubscribe func(), ok bool) {
	r.mu.Lock()
	defer r.mu.Unlock()
	a := r.active
	if a == nil || a.id != id {
		return nil, nil, nil, false
	}
	c := make(chan Sample, 64)
	a.subs[c] = struct{}{}
	past = append([]Sample(nil), a.samples...)
	unsubscribe = func() {
		r.mu.Lock()
		defer r.mu.Unlock()
		if _, found := a.subs[c]; found {
			delete(a.subs, c)
			close(c)
		}
	}
	return past, c, unsubscribe, true
}

// Shutdown stops the active run, if any, and waits for it to be persisted.
func (r *Runner) Shutdown() {
	r.mu.Lock()
	a := r.active
	r.mu.Unlock()
	if a == nil {
		return
	}
	a.stopped.Store(true)
	a.cancel()
	<-a.done
}
```

```go
var staticFS embed.FS

// RunDetail is a run plus its per-second samples.
type RunDetail struct {
	Run     Run      `json:"run"`
	Samples []Sample `json:"samples"`
}

type server struct {
	store  *Store
	runner *Runner
}

func NewServer(store *Store, runner *Runner) http.Handler {
	s := &server{store: store, runner: runner}
	mux := http.NewServeMux()
	mux.HandleFunc("POST /api/runs", s.startRun)
	mux.HandleFunc("GET /api/runs", s.listRuns)
	mux.HandleFunc("GET /api/runs/{id}", s.getRun)
	mux.HandleFunc("GET /api/runs/{id}/events", s.events)
	mux.HandleFunc("DELETE /api/runs/{id}", s.stopRun)
	mux.HandleFunc("GET /api/compare", s.compare)

	static, err := fs.Sub(staticFS, "static")
	if err != nil {
		panic(err) // embedded path is fixed at compile time
	}
	mux.Handle("GET /", http.FileServerFS(static))
	return sameOriginOnly(mux)
}

// sameOriginOnly blocks cross-site requests from a browser: any mutating
// request carrying an Origin must match the Host it was sent to.
func sameOriginOnly(next http.Handler) http.Handler {
	return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		if r.Method != http.MethodGet && r.Method != http.MethodHead {
			if origin := r.Header.Get("Origin"); origin != "" {
				u, err := url.Parse(origin)
				if err != nil || u.Host != r.Host {
					writeError(w, http.StatusForbidden, "cross-origin request blocked", "")
					return
				}
			}
		}
		next.ServeHTTP(w, r)
	})
}

func writeJSON(w http.ResponseWriter, status int, v any) {
	w.Header().Set("Content-Type", "application/json")
	w.WriteHeader(status)
	_ = json.NewEncoder(w).Encode(v)
}

func writeError(w http.ResponseWriter, status int, msg, field string) {
	body := map[string]string{"error": msg}
	if field != "" {
		body["field"] = field
	}
	writeJSON(w, status, body)
}

func parseID(w http.ResponseWriter, raw string) (int64, bool) {
	id, err := strconv.ParseInt(raw, 10, 64)
	if err != nil || id <= 0 {
		writeError(w, http.StatusBadRequest, "invalid run id", "")
		return 0, false
	}
	return id, true
}

func (s *server) startRun(w http.ResponseWriter, r *http.Request) {
	if !strings.HasPrefix(r.Header.Get("Content-Type"), "application/json") {
		writeError(w, http.StatusUnsupportedMediaType, "Content-Type must be application/json", "")
		return
	}
	var req RunRequest
	dec := json.NewDecoder(http.MaxBytesReader(w, r.Body, 1<<20))
	dec.DisallowUnknownFields()
	if err := dec.Decode(&req); err != nil {
		writeError(w, http.StatusBadRequest, "invalid JSON body: "+err.Error(), "")
		return
	}

	id, err := s.runner.Start(req)
	var ve *ValidationError
	switch {
	case errors.As(err, &ve):
		writeError(w, http.StatusBadRequest, ve.Message, ve.Field)
	case errors.Is(err, ErrBusy):
		writeError(w, http.StatusConflict, err.Error(), "")
	case err != nil:
		writeError(w, http.StatusInternalServerError, err.Error(), "")
	default:
		writeJSON(w, http.StatusCreated, map[string]int64{"id": id})
	}
}

func (s *server) listRuns(w http.ResponseWriter, _ *http.Request) {
	runs, err := s.store.ListRuns()
	if err != nil {
		writeError(w, http.StatusInternalServerError, err.Error(), "")
		return
	}
	writeJSON(w, http.StatusOK, runs)
}

func (s *server) detail(id int64) (RunDetail, error) {
	run, err := s.store.GetRun(id)
	if err != nil {
		return RunDetail{}, err
	}
	samples, err := s.store.Samples(id)
	if err != nil {
		return RunDetail{}, err
	}
	return RunDetail{Run: run, Samples: samples}, nil
}

func (s *server) writeDetailError(w http.ResponseWriter, err error) {
	if errors.Is(err, ErrNotFound) {
		writeError(w, http.StatusNotFound, err.Error(), "")
		return
	}
	writeError(w, http.StatusInternalServerError, err.Error(), "")
}

func (s *server) getRun(w http.ResponseWriter, r *http.Request) {
	id, ok := parseID(w, r.PathValue("id"))
	if !ok {
		return
	}
	d, err := s.detail(id)
	if err != nil {
		s.writeDetailError(w, err)
		return
	}
	writeJSON(w, http.StatusOK, d)
}

func (s *server) stopRun(w http.ResponseWriter, r *http.Request) {
	id, ok := parseID(w, r.PathValue("id"))
	if !ok {
		return
	}
	if !s.runner.Stop(id) {
		writeError(w, http.StatusNotFound, "no active run with that id", "")
		return
	}
	w.WriteHeader(http.StatusNoContent)
}

func (s *server) compare(w http.ResponseWriter, r *http.Request) {
	a, okA := parseID(w, r.URL.Query().Get("a"))
	if !okA {
		return
	}
	b, okB := parseID(w, r.URL.Query().Get("b"))
	if !okB {
		return
	}
	da, err := s.detail(a)
	if err != nil {
		s.writeDetailError(w, err)
		return
	}
	db, err := s.detail(b)
	if err != nil {
		s.writeDetailError(w, err)
		return
	}
	writeJSON(w, http.StatusOK, map[string]RunDetail{"a": da, "b": db})
}

func sendEvent(w http.ResponseWriter, f http.Flusher, event string, v any) {
	b, err := json.Marshal(v)
	if err != nil {
		return
	}
	fmt.Fprintf(w, "event: %s\ndata: %s\n\n", event, b)
	f.Flush()
}

func (s *server) events(w http.ResponseWriter, r *http.Request) {
	id, ok := parseID(w, r.PathValue("id"))
	if !ok {
		return
	}
	flusher, ok := w.(http.Flusher)
	if !ok {
		writeError(w, http.StatusInternalServerError, "streaming unsupported", "")
		return
	}

	past, ch, unsubscribe, active := s.runner.Subscribe(id)
	if !active {
		d, err := s.detail(id)
		if err != nil {
			s.writeDetailError(w, err)
			return
		}
		startStream(w)
		for _, smp := range d.Samples {
			sendEvent(w, flusher, "sample", smp)
		}
		sendEvent(w, flusher, "done", d.Run)
		return
	}
	defer unsubscribe()

	startStream(w)
	for _, smp := range past {
		sendEvent(w, flusher, "sample", smp)
	}
	flusher.Flush()
	for {
		select {
		case <-r.Context().Done():
			return
		case smp, open := <-ch:
			if !open {
				if run, err := s.store.GetRun(id); err == nil {
					sendEvent(w, flusher, "done", run)
				}
				return
			}
			sendEvent(w, flusher, "sample", smp)
		}
	}
}

func startStream(w http.ResponseWriter) {
	h := w.Header()
	h.Set("Content-Type", "text/event-stream")
	h.Set("Cache-Control", "no-cache")
	h.Set("Connection", "keep-alive")
	w.WriteHeader(http.StatusOK)
}
```

```go
		dbPath := webDB
		if dbPath == "" {
			home, err := os.UserHomeDir()
			if err != nil {
				return fmt.Errorf("cannot locate home directory (use --db): %w", err)
			}
			dbPath = filepath.Join(home, ".yahba", "history.db")

		}

		if err != nil {
			return fmt.Errorf("opening history database: %w", err)
		}
		defer store.Close()
		if err := store.MarkAbandoned(); err != nil {
			return err
		}

		runner := web.NewRunner(store)
		srv := &http.Server{
			Addr:              webAddr,
			Handler:           web.NewServer(store, runner),
			ReadHeaderTimeout: 10 * time.Second,
		}

		ctx, stop := signal.NotifyContext(context.Background(), os.Interrupt, syscall.SIGTERM)
		defer stop()

		errCh := make(chan error, 1)
		go func() { errCh <- srv.ListenAndServe() }()
		fmt.Printf("YAHBA web UI listening on http://%s (history: %s)\n", webAddr, dbPath)

		select {
		case err := <-errCh:
			return err
		case <-ctx.Done():
		}

		runner.Shutdown()
		shutdownCtx, cancel := context.WithTimeout(context.Background(), 3*time.Second)
		defer cancel()
		return srv.Shutdown(shutdownCtx)
	},
}
```

```ts
export function formatMs(ms: number): string {
  if (ms >= 1000) return `${(ms / 1000).toFixed(2)} s`;
  if (ms >= 1) return `${ms.toFixed(1)} ms`;
  return `${(ms * 1000).toFixed(0)} µs`;
}

export function formatDuration(ms: number): string {
  if (ms < 60000) return `${(ms / 1000).toFixed(1)}s`;
  const totalSeconds = Math.floor(ms / 1000);
  const minutes = Math.floor(totalSeconds / 60);
  const seconds = String(totalSeconds % 60).padStart(2, '0');
  return `${minutes}m ${seconds}s`;
}

export function formatBytes(n: number): string {
```

```ts
export interface Delta {
  label: string;
  a: number;
  b: number;
  diff: number;
  pct: number | null;
  lowerIsBetter: boolean;
  unit: 'ms' | 'rps' | 'count';

  { label: 'Req/s', pick: (s) => s.rps, lowerIsBetter: false, unit: 'rps' },
  { label: 'Avg', pick: (s) => s.avg_ms, lowerIsBetter: true, unit: 'ms' },
  { label: 'p50', pick: (s) => s.p50_ms, lowerIsBetter: true, unit: 'ms' },
  { label: 'p95', pick: (s) => s.p95_ms, lowerIsBetter: true, unit: 'ms' },
  { label: 'p99', pick: (s) => s.p99_ms, lowerIsBetter: true, unit: 'ms' },
  { label: 'Max', pick: (s) => s.max_ms, lowerIsBetter: true, unit: 'ms' },
  { label: 'Failures', pick: (s) => s.failures, lowerIsBetter: true, unit: 'count' },
];

export function compareSummaries(a: Summary, b: Summary): Delta[] {
  return rows.map(({ label, pick, lowerIsBetter, unit }) => {
    const av = pick(a);
    const bv = pick(b);
    const diff = bv - av;
    return { label, a: av, b: bv, diff, pct: av === 0 ? null : (diff / av) * 100, lowerIsBetter, unit };
  });
}

export function alignSeries(
  a: Sample[],
  b: Sample[],
  pick: (s: Sample) => number,
): { xs: number[]; a: (number | null)[]; b: (number | null)[] } {
  const xs = [...new Set([...a, ...b].map((s) => s.t))].sort((x, y) => x - y);
  const ma = new Map(a.map((s) => [s.t, pick(s)]));
  const mb = new Map(b.map((s) => [s.t, pick(s)]));
  return { xs, a: xs.map((x) => ma.get(x) ?? null), b: xs.map((x) => mb.get(x) ?? null) };
}
```

```ts
type Child = Node | string;

// h builds an element. String children become text nodes, so user-supplied
// values (URLs, error messages) can never inject markup.
export function h<K extends keyof HTMLElementTagNameMap>(
  tag: K,
  props: Record<string, unknown> = {},
  ...children: Child[]
): HTMLElementTagNameMap[K] {
  const el = document.createElement(tag);
  for (const [key, value] of Object.entries(props)) {
    if (key.startsWith('on') && typeof value === 'function') {
      el.addEventListener(key.slice(2).toLowerCase(), value as EventListener);
    } else if (key === 'class') {
      el.className = String(value);
    } else if (value === true) {
      el.setAttribute(key, '');
    } else if (value !== false && value !== null && value !== undefined) {
      el.setAttribute(key, String(value));
    }
  }
  el.append(...children);
  return el;
}
```

```ts
import type { Run, RunDetail, RunRequest, Sample } from './types';

export class ApiError extends Error {
  constructor(
    public status: number,
    message: string,
    public field?: string,
  ) {
    super(message);
  }
}

async function parse<T>(res: Response): Promise<T> {
  if (!res.ok) {
    const body = (await res.json().catch(() => ({}))) as { error?: string; field?: string };
    throw new ApiError(res.status, body.error ?? res.statusText, body.field);
  }
  return (await res.json()) as T;
}

export const startRun = (req: RunRequest) =>
  fetch('/api/runs', {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify(req),
  }).then((r) => parse<{ id: number }>(r));

export const listRuns = () => fetch('/api/runs').then((r) => parse<Run[]>(r));

export const getRun = (id: number) => fetch(`/api/runs/${id}`).then((r) => parse<RunDetail>(r));

export const stopRun = async (id: number): Promise<void> => {
  const res = await fetch(`/api/runs/${id}`, { method: 'DELETE' });
  if (!res.ok) await parse<unknown>(res);
};

export const getCompare = (a: number, b: number) =>
  fetch(`/api/compare?a=${a}&b=${b}`).then((r) => parse<{ a: RunDetail; b: RunDetail }>(r));

// streamRun replays past samples then follows live ones. EventSource reconnects
// on its own and the server replays, so callers must treat samples as idempotent by `t`.
export function streamRun(id: number, onSample: (s: Sample) => void, onDone: (run: Run) => void): () => void {
  const es = new EventSource(`/api/runs/${id}/events`);
  es.addEventListener('sample', (e) => onSample(JSON.parse((e as MessageEvent).data) as Sample));
  es.addEventListener('done', (e) => {
    es.close();
    onDone(JSON.parse((e as MessageEvent).data) as Run);
  });
  return () => es.close();
}
```

```ts
const fmt = (v: number, unit: Delta['unit']) =>
  unit === 'ms' ? formatMs(v) : unit === 'rps' ? v.toFixed(1) : String(v);

export function renderCompare(root: HTMLElement, a: number, b: number): () => void {
  const errorEl = h('p', { class: 'error' });
  const tableEl = h('div');
  const rpsEl = h('div', { class: 'chart' });
  const latEl = h('div', { class: 'chart' });
  root.append(h('h2', {}, `Run #${a} vs #${b}`), errorEl, tableEl, rpsEl, latEl);

  const rps = lineChart(rpsEl, 'Requests per second', [

    { label: `#${b}`, stroke: '#dc2626', dash: [6, 4] },

  const lat = lineChart(latEl, 'p95 latency', [
    { label: `#${a}`, stroke: '#2563eb' },
    { label: `#${b}`, stroke: '#dc2626', dash: [6, 4] },
  ], 'ms');

  let disposed = false;
  getCompare(a, b)
    .then(({ a: da, b: db }) => {
      if (disposed) return;
      const r = alignSeries(da.samples, db.samples, (s) => s.count);
      rps.update(r.xs, [r.a, r.b]);
      const l = alignSeries(da.samples, db.samples, (s) => s.p95_ms);
      lat.update(l.xs, [l.a, l.b]);

          h('td', {}, fmt(d.a, d.unit)),
          h('td', {}, fmt(d.b, d.unit)),
          h('td', { class: better }, pct),
        );
      });
      tableEl.replaceChildren(
        h(
          'table',
          {},
          h('thead', {}, h('tr', {}, ...['', `#${a}`, `#${b}`, 'Change'].map((c) => h('th', {}, c)))),
          h('tbody', {}, ...rows),
        ),
      );
    })
    .catch((e: Error) => {
```
